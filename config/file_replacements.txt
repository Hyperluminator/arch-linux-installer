# Format: `/origin/directory > /target/directory`, where the origin directory is relative to the installer script's directory and the target directory is an absolute path.

# Example:
# dotfiles/.bashrc > /etc/skel/.bashrc
# dotfiles/.config/gtk-4.0/settings.ini > /etc/skel/.config/gtk-4.0/settings.ini
